Initialize Map with Free OpenStreetMap (No API Key Required) function initMap() { map = L.map('map', { center: [13.736717, 100.523186], zoom: 6, minZoom: 5, maxZoom: 10 }); // ใช้ OpenStreetMap มาตรฐาน ใช้งานได้ฟรีตลอดเวลา ไม่ติด API Key L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', { maxZoom: 19, attribution: '© OpenStreetMap contributors' }).addTo(map); addLegend(); } function resetMapView() { if (map) map.setView([13.736717, 100.523186], 6); } // 2. Color Scales function getColor(d) { return d > 120 ? '#800026' : d > 80 ? '#BD0026' : d > 50 ? '#E31A1C' : d > 30 ? '#FC4E2A' : d > 15 ? '#FD8D3C' : d > 5 ? '#FEB24C' : d > 0 ? '#FED976' : '#f0f0f0'; } function addLegend() { const legend = L.control({ position: 'bottomright' }); legend.onAdd = function () { const div = L.DomUtil.create('div', 'legend'); const grades = [0, 5, 15, 30, 50, 80, 120]; div.innerHTML = 'จำนวนผู้เสียชีวิต (ราย)
'; for (let i = 0; i < grades.length; i++) { div.innerHTML += ' ' + grades[i] + (grades[i + 1] ? '–' + grades[i + 1] + '
' : '+'); } return div; }; legend.addTo(map); } // 3. Load Thailand 77 Provinces GeoJSON (มีระบบ Fallback สำรองหากเซิร์ฟเวอร์ใดล่ม) async function loadThailandGeoJSON() { const geojsonSources = [ 'https://raw.githubusercontent.com/chingchai/OpenGISData-Thailand/master/provinces.geojson', 'https://raw.githubusercontent.com/apisit/thailand.json/master/thailandWithName.json', 'https://raw.githubusercontent.com/cvibhagool/thailand-map/master/thailand-provinces.geojson' ]; for (const url of geojsonSources) { try { const res = await fetch(url); if (res.ok) { thaiGeoJson = await res.json(); console.log('โหลด GeoJSON สำเร็จจาก:', url); renderMapLayers(); return; } } catch (e) { console.warn('กำลังลองโหลดจากแหล่งสำรองถัดไป...'); } } } // ดึงชื่อจังหวัดจาก GeoJSON ให้ตรงกับชื่อในไฟล์ CSV ภาษาไทย function getProvinceName(props) { return (props.pro_th || props.name || props.NAME_1 || props.NL_NAME_1 || props.dname || '').replace(/^(จ\.|จังหวัด)/, '').trim(); } function renderMapLayers() { if (!thaiGeoJson || !map) return; if (geojsonLayer) map.removeLayer(geojsonLayer); // รวบรวมสถิติตามจังหวัดจากข้อมูลจริง const provCounts = {}; filteredData.forEach(row => { const p = (row.Dead_Prov || '').replace(/^(จ\.|จังหวัด)/, '').trim(); if (p) provCounts[p] = (provCounts[p] || 0) + 1; }); geojsonLayer = L.geoJson(thaiGeoJson, { style: function(feature) { const name = getProvinceName(feature.properties); const count = provCounts[name] || 0; return { fillColor: getColor(count), weight: 1.2, opacity: 1, color: '#ffffff', dashArray: '2', fillOpacity: 0.75 }; }, onEachFeature: function(feature, layer) { const name = getProvinceName(feature.properties); const count = provCounts[name] || 0; // Tooltip เมื่อชี้เมาส์ layer.bindTooltip(`จังหวัด${name}
ผู้เสียชีวิต: ${count.toLocaleString()} ราย`, { sticky: true, className: 'custom-tooltip' }); // เหตุการณ์คลิกและ Hover layer.on({ mouseover: function(e) { const l = e.target; l.setStyle({ weight: 3, color: '#ff7a00', fillOpacity: 0.9 }); }, mouseout: function(e) { geojsonLayer.resetStyle(e.target); }, click: function() { showProvinceDetail(name, count); } }); } }).addTo(map); }
